const identity = reactive({ email:'test@example.com', companyName:'远舟跨境演示客户', contactName:'陈经理', customerType:'ecommerce_seller' })
const showDetails = ref(false)
const quoteLoading = ref(false)
const identityLoading = ref(false)
const rematchLoading = ref(false)
const sessionId = ref('')
const inquiryId = ref('')
const recommendations = ref<any[]>([])
const selectedProductId = ref('')
const customerId = ref('')
const opportunityId = ref('')
const taskId = ref('')
const phase = ref<'need'|'identity'|'agent'>('need')

const relatedTasks = computed(() => (state.value?.tasks || []).filter(task => task.targetId === customerId.value || task.id === taskId.value).slice(0,4))
const activeTask = computed(() => relatedTasks.value.find(task => ['queued','running','waiting'].includes(task.status)))
const latestTask = computed(() => activeTask.value || relatedTasks.value[0])
const productById = (id:string) => state.value?.products.find(product => product.id === id)

onMounted(() => { void refresh({ quiet:true }) })

async function quote() {
  quoteLoading.value = true
  try {
    const result = await $fetch<any>('/api/website/quote', { method:'POST', body:{ ...form, sessionId:sessionId.value || undefined, inquiryId:inquiryId.value || undefined } })
    sessionId.value = result.sessionId; inquiryId.value = result.inquiryId; recommendations.value = result.recommendations; selectedProductId.value = result.recommendations[0]?.productId || ''; phase.value = 'identity'
    Message.success('已匹配已发布产品，请留资保存本次询价')
    requestAnimationFrame(() => document.getElementById('quote-results')?.scrollIntoView({ behavior:'smooth', block:'start' }))
  } catch (error:any) { Message.error(error?.data?.statusMessage || '查询失败，请检查输入后重试') }
  finally { quoteLoading.value = false }
}
async function identify() {
  if (!selectedProductId.value) return Message.warning('请选择一个产品方案')
  identityLoading.value = true
  try {
    const result = await $fetch<any>('/api/website/identity', { method:'POST', body:{ inquiryId:inquiryId.value, ...identity, selectedProductId:selectedProductId.value } })
    customerId.value = result.customerId; opportunityId.value = result.opportunityId; taskId.value = result.task?.task?.id || ''; phase.value = 'agent'
    Message.info('客户档案已建立，Agent 已开始分析；请勿重复提交')
    await refresh({ quiet:true })
  } catch (error:any) { Message.error(error?.data?.statusMessage || '身份留资失败，请稍后重试') }
  finally { identityLoading.value = false }
}
async function rematch() {
  if (!inquiryId.value) return quote()
  rematchLoading.value = true
  try {
    const result = await $fetch<any>('/api/website/rematch', { method:'POST', body:{ inquiryId:inquiryId.value, ...form } })
    recommendations.value = result.recommendations; selectedProductId.value = result.recommendations[0]?.productId || selectedProductId.value; if (result.task?.task?.id) taskId.value = result.task.task.id
    Message.info(customerId.value ? '询价已更新，Agent 已开始重新匹配' : '询价已更新，推荐结果已刷新')
    await refresh({ quiet:true })
  } catch (error:any) { Message.error(error?.data?.statusMessage || '重新匹配失败') }
  finally { rematchLoading.value = false }
}
function resetFlow() { phase.value='need';sessionId.value='';inquiryId.value='';recommendations.value=[];selectedProductId.value='';customerId.value='';opportunityId.value='';taskId.value='';window.scrollTo({top:0,behavior:'smooth'}) }
